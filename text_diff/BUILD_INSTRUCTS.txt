# ======================================================
# Android Build Instructions for text_diff UniFFI
# ======================================================

# --- PRE-BUILD (once) ---
# 1. Install cargo-ndk:
#    cargo install cargo-ndk
# 2. Add ARM64 target:
#    rustup target add aarch64-linux-android

# --- BUILD STEPS (run from inside the text_diff folder) ---

# Step 0: Clean previous output to prevent ghost dependencies (like libwasm_streams)
rm -rf ./android_output_folder

# Step 1: Build native library for arm64
# Added --lib to ensure only the shared library is compiled for Android
cargo ndk -o ./android_output_folder/jniLibs -t arm64-v8a build --release --lib

# Step 2: Generate Kotlin glue code (force host target)
cargo run --target "$(rustc -vV | sed -n 's/host: //p')" --bin uniffi-bindgen -- generate \
    --library ./android_output_folder/jniLibs/arm64-v8a/libtext_diff.so \
    --language kotlin \
    --out-dir ./android_output_folder/kotlin \
    --config ./uniffi.toml

# --- RESULT ---
# The Kotlin file will be named `text_diff.kt` and located under
# android_output_folder/kotlin/rustlib/text_diff/.
# Copy the `jniLibs` and `kotlin` directories into your Android project.

# --- NOTES ---
# - Only the `diff_logic` module is compiled for Android (other modules are
#   excluded via cfg(target_arch = "wasm32") in lib.rs).
# - To build for other architectures, repeat Step 1 with: -t armeabi-v7a, -t x86, -t x86_64
# - Re-run both steps after any Rust code changes.
