# ======================================================
# Android Build Instructions for LoveLetter UniFFI
# ======================================================

# --- PRE-BUILD (once) ---
# 1. Install cargo-ndk:
#    cargo install cargo-ndk
# 2. Add ARM64 target:
#    rustup target add aarch64-linux-android

# --- BUILD STEPS ---
# 3. Build native library for arm64
cargo ndk -o ./android_output_folder/jniLibs -t arm64-v8a build --release

# 4. Generate Kotlin glue code
cargo run --bin uniffi-bindgen -- generate \
    --library ./android_output_folder/jniLibs/arm64-v8a/liblove_letter.so \
    --language kotlin \
    --out-dir ./android_output_folder/kotlin \
    --config ./uniffi.toml

# --- RESULT ---
# The Kotlin file will be named `love_letter.kt` (same as your crate name)
# and located under android_output_folder/kotlin/rustlib/love_letter/.
# You can copy the `jniLibs` and `kotlin` directories into your Android project.

# --- NOTES ---
# - To change the Kotlin file name, rename your crate in Cargo.toml.
# - To build for other architectures, repeat step 3 with: -t armeabi-v7a, -t x86, -t x86_64
# - Re-run steps 3 and 4 after any Rust code changes.
