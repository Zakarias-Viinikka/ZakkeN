(z_db/) folder is a manual copy of db_wrapper/src/web_output in:
https://github.com/Zakarias-Viinikka/z_db

#the script.sh that automatically updates uses z_db env path i've set, so it doesn't work unless you modify it or set that enviroment yourself.
#it also requires jq which you can check if you have by running:
#
which jq
#
#

###this will work on any machine since it gets the dependency from github.
### in order use it you copy the thing below and run it in terminal

REPO="Zakarias-Viinikka/z_db"; SUBPATH="db_wrapper/src/web_output"; DEST="z_db"; HASH_FILE=".last_web_output_hash"; LATEST=$(curl -s "https://api.github.com/repos/$REPO/commits?path=$SUBPATH&per_page=1" | jq -r '.[0].sha'); OLD=$(cat "$HASH_FILE" 2>/dev/null || echo ""); if [ "$LATEST" = "$OLD" ]; then echo "web_output unchanged, skipping copy"; else TMP_DIR=$(mktemp -d); git clone --depth 1 --filter=blob:none --sparse "https://github.com/$REPO.git" "$TMP_DIR" > /dev/null 2>&1; cd "$TMP_DIR"; git sparse-checkout set "$SUBPATH"; cd - > /dev/null; cp -r "$TMP_DIR/$SUBPATH/." "$DEST"; echo "$LATEST" > "$HASH_FILE"; rm -rf "$TMP_DIR"; echo "web_output updated"; fi
